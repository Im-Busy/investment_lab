# Integration test - detect patterns on random data
uv run python -c "
import numpy as np, pandas as pd
np.random.seed(42)
n = 300
close = 100 + np.cumsum(np.random.randn(n) * 0.5)
high = close + np.abs(np.random.randn(n))
low = close - np.abs(np.random.randn(n))
df = pd.DataFrame({'Open': np.roll(close,1), 'High': high, 'Low': low, 'Close': close, 'Volume': np.random.randint(5000,20000,n).astype(float)})
from src.patterns.fmz import AlphaBeast, MultiFactorTrend, EMAMACDHF, AdaptiveBollinger, AIVolatilityBreakout, MomentumZigZag
for d in [AlphaBeast(), MultiFactorTrend(), EMAMACDHF(), AdaptiveBollinger(), AIVolatilityBreakout(), MomentumZigZag()]:
    found = any(d.detect(df, i).detected for i in range(200, n))
    print(f'{d.name}: {\"detected\" if found else \"no detection (random data)\"}')
"
