//@version=5
indicator("Asian Sweep + MSS + IFVG + HTF Bias [V5]", overlay=true, shorttitle="AsianSweep-MSS-IFVG", precision=5)

// ———————————————————— 1. 输入设置 ————————————————————
// 亚洲时段
var string tz = "GMT+0"
asia_start = input.time(timestamp("00:00", "GMT+8"), "亚洲盘开始", inline="1")
asia_end   = input.time(timestamp("08:00", "GMT+8"), "结束", inline="1")

// 枢轴长度（判断结构）
pivot_len = input.int(5, "枢轴长度", minval=1)

// FVG / IFVG
fvg_min_size = input.float(0.0, "最小FVG大小", step=0.1)
lookback_fvg = input.int(20, "FVG回看K线数")

// HTF 偏向
htf = input.string("240", "HTF偏向周期", options=["60", "240", "D"])

// 画图开关
show_asian_box = input(true, "显示亚洲区间")
show_sweep    = input(true, "显示扫盘标记")
show_mss      = input(true, "显示MSS结构")
show_fvg      = input(true, "显示FVG/IFVG")

// ———————————————————— 2. 亚洲区间 ————————————————————
var float asia_high = na
var float asia_low  = na
var bool in_asia = ta.timewindow(asia_start, asia_end)

if in_asia and not in_asia[1]
    asia_high := high
    asia_low  := low
if in_asia
    asia_high := math.max(asia_high, high)
    asia_low  := math.min(asia_low, low)

// ———————————————————— 3. 流动性扫盘 Sweep ————————————————————
var bool swept_up = false
var bool swept_dn = false
var float sweep_hi = na
var float sweep_lo = na

buffer = input.float(0.0, "扫盘缓冲点数", step=0.1)

broke_hi = high > asia_high + buffer and not in_asia
broke_lo = low  < asia_low  - buffer and not in_asia

if (broke_hi or broke_lo) and not swept_up and not swept_dn
    swept_up := broke_hi
    swept_dn := broke_lo
    sweep_hi := high
    sweep_lo := low

// ———————————————————— 4. 枢轴高低点 ————————————————————
pl = ta.pivotlow(pivot_len, pivot_len)
ph = ta.pivothigh(pivot_len, pivot_len)
var float last_ph = na
var float last_pl = na
if ph
    last_ph := ph
if pl
    last_pl := pl

// ———————————————————— 5. MSS 结构转变 ————————————————————
bull_mss = ta.close > last_ph and swept_dn
bear_mss = ta.close < last_pl and swept_up

// ———————————————————— 6. FVG & IFVG ————————————————————
// 多头FVG：中间K线低点 > 前一根高点
bull_fvg = low[1] > high[2] and high[1] < low[0]
// 空头FVG：中间K线高点 < 前一根低点
bear_fvg = high[1] < low[2] and low[1] > high[0]

// IFVG = 被价格穿过并反转测试的FVG
bull_ifvg = bull_fvg and ta.close < low[1] and ta.close[1] > high[2]
bear_ifvg = bear_fvg and ta.close > high[1] and ta.close[1] < low[2]

// ———————————————————— 7. HTF 4H 偏向 ————————————————————
htf_close = request.security(syminfo.tickerid, htf, close, barmerge.gaps_on, barmerge.lookahead_off)
htf_ma = ta.sma(htf_close, 20)
htf_bull = htf_close > htf_ma
htf_bear = htf_close < htf_ma

// ———————————————————— 8. 交易信号 ————————————————————
long  = bull_mss and htf_bull and bull_ifvg
short = bear_mss and htf_bear and bear_ifvg

// ———————————————————— 9. 画图 ————————————————————
// 亚洲区间
if show_asian_box
    box.new(asia_start, asia_high, asia_end, asia_low, border_color=color.gray, bgcolor=color.new(color.gray, 90), border_width=1)

// 扫盘
if show_sweep and swept_up
    label.new(bar_index, high, "SWEEP HI", color=color.new(color.red, 80), textcolor=color.white, style=label.style_label_down)
if show_sweep and swept_dn
    label.new(bar_index, low, "SWEEP LO", color=color.new(color.green, 80), textcolor=color.white, style=label.style_label_up)

// MSS
if show_mss and bull_mss
    plotshape(series=bull_mss, title="Bull MSS", location=location.belowbar, color=color.green, style=shape.labelup, text="MSS LONG")
if show_mss and bear_mss
    plotshape(series=bear_mss, title="Bear MSS", location=location.abovebar, color=color.red, style=shape.labeldown, text="MSS SHORT")

// FVG / IFVG
if show_fvg and bull_ifvg
    plot(low[1], "IFVG Bull", color=color.new(color.green, 50), linewidth=3)
if show_fvg and bear_ifvg
    plot(high[1], "IFVG Bear", color=color.new(color.red, 50), linewidth=3)

// 最终信号
plotshape(long, "LONG", location.belowbar, color=color.green, style=shape.arrowup, size=size.small)
plotshape(short, "SHORT", location.abovebar, color=color.red, style=shape.arrowdown, size=size.small)
