//@version=5
strategy("Asian Sweep + IFVG Strategy [Algo]", shorttitle="AS+IFVG", overlay=true, max_lines_count=500, max_boxes_count=500, max_labels_count=500, calc_on_order_fills=true, initial_capital=10000, default_qty_type=strategy.percent_of_equity, default_qty_value=1, currency=currency.USD)

// =============================================================================
// 1. INPUTS
// =============================================================================

// --- Session & Timeframes ---
grp_time = "Time Settings"
asia_sess     = input.session("1900-0000", "Asian Session (Exch. Time)", group=grp_time)
tf_htf        = input.timeframe("240", "HTF Bias Timeframe", group=grp_time)
use_daily     = input.bool(false, "Use Daily Bias Confirmation?", group=grp_time)
tf_daily      = input.timeframe("D", "Daily Timeframe", group=grp_time)

// --- Structure & Sweep ---
grp_struct = "Structure & Sweep"
sw_left       = input.int(3, "Pivot Left Bars", group=grp_struct)
sw_right      = input.int(3, "Pivot Right Bars", group=grp_struct)
sweep_mode    = input.string("Wick", "Sweep Mode", options=["Wick", "Close"], group=grp_struct)
sweep_buff    = input.float(0.0, "Sweep Buffer (Pips)", group=grp_struct)
mss_len       = input.int(5, "MSS Pivot Lookback", group=grp_struct)

// --- FVG / IFVG ---
grp_fvg = "FVG & IFVG"
fvg_threshold = input.float(0.5, "Min FVG Size (ATR mult)", group=grp_fvg)
fvg_atr_len   = input.int(14, "ATR Length for FVG", group=grp_fvg)
show_fvgs     = input.bool(true, "Show FVGs", group=grp_fvg)
show_ifvgs    = input.bool(true, "Show IFVGs (Flips)", group=grp_fvg)
max_lookback  = input.int(50, "Max Zone Lookback Bars", group=grp_fvg)

// --- Strategy ---
grp_strat = "Strategy Execution"
use_strat     = input.bool(false, "Enable Strategy Mode (Backtest)", group=grp_strat)
req_htf_bias  = input.bool(true, "Require HTF Bias Match", group=grp_strat)
req_ifvg      = input.bool(true, "Require Valid FVG/IFVG for Entry", group=grp_strat)
entry_type    = input.string("Limit", "Entry Type", options=["Market", "Limit"], group=grp_strat)
limit_level   = input.string("Zone Start", "Limit Level", options=["Zone Start", "Zone Mid", "Zone End"], group=grp_strat)
risk_per_tr   = input.float(1.0, "Risk % Per Trade", group=grp_strat)
sl_padding    = input.float(1.0, "SL Padding (ATR mult)", group=grp_strat)
rr_target     = input.float(2.0, "Risk:Reward Target", group=grp_strat)

// --- Visuals ---
grp_vis = "Visuals"
show_asia     = input.bool(true, "Show Asian Box", group=grp_vis)
col_asia      = input.color(color.new(color.blue, 90), "Asian Range Color", group=grp_vis)
col_bull      = input.color(color.new(color.green, 60), "Bullish Zone", group=grp_vis)
col_bear      = input.color(color.new(color.red, 60), "Bearish Zone", group=grp_vis)
show_table    = input.bool(true, "Show Status Panel", group=grp_vis)

// =============================================================================
// 2. DEFINITIONS & UTILS
// =============================================================================

// ATR for calculations
atrVal = ta.atr(fvg_atr_len)
pipSize = syminfo.mintick * 10

// User Defined Type for Zones (FVG/IFVG)
type Zone
    box     id
    float   top
    float   btm
    bool    isBull
    bool    isMitigated
    bool    isInverted // IFVG
    int     barIndex
    bool    active

var Zone[] zones = array.new<Zone>()

// Clean up old boxes/zones
f_manage_zones() =>
    if array.size(zones) > 0
        for i = array.size(zones) - 1 to 0
            z = array.get(zones, i)
            // Remove if too old or fully invalidated/consumed (logic simplified for perf)
            if bar_index - z.barIndex > max_lookback * 4
                box.delete(z.id)
                array.remove(zones, i)

// =============================================================================
// 3. ASIAN RANGE LOGIC
// =============================================================================

var float asia_high = na
var float asia_low  = na
var box asia_box    = na
var bool session_active = false

bool in_session = not na(time(timeframe.period, asia_sess))

// Detect Session Start/End
if in_session and not in_session[1]
    // Start of session
    asia_high := high
    asia_low  := low
    session_active := true
    if show_asia
        asia_box := box.new(bar_index, asia_high, bar_index, asia_low, border_color=na, bgcolor=col_asia)

if in_session
    // Update High/Low during session
    asia_high := math.max(asia_high, high)
    asia_low  := math.min(asia_low, low)
    if show_asia and not na(asia_box)
        box.set_top(asia_box, asia_high)
        box.set_bottom(asia_box, asia_low)
        box.set_right(asia_box, bar_index)
else
    session_active := false

// =============================================================================
// 4. SWEEP DETECTION
// =============================================================================

var string swept_side = "None" // "High", "Low", "None"
var bool sweep_confirmed = false

// Reset sweep logic at start of new session
if in_session and not in_session[1]
    swept_side := "None"
    sweep_confirmed := false

// Detect Sweep after session
if not in_session and not sweep_confirmed and not na(asia_high)
    // Bullish Sweep (Sweep Low)
    float low_check = sweep_mode == "Wick" ? low : close
    if low_check < (asia_low - sweep_buff * syminfo.mintick)
        swept_side := "Low"
        sweep_confirmed := true
        alert("Asian Low Swept", alert.freq_once_per_bar)
        if show_asia
            label.new(bar_index, low, "Sweep", style=label.style_label_up, color=color.green, size=size.tiny)

    // Bearish Sweep (Sweep High)
    float high_check = sweep_mode == "Wick" ? high : close
    if high_check > (asia_high + sweep_buff * syminfo.mintick)
        swept_side := "High"
        sweep_confirmed := true
        alert("Asian High Swept", alert.freq_once_per_bar)
        if show_asia
            label.new(bar_index, high, "Sweep", style=label.style_label_down, color=color.red, size=size.tiny)

// =============================================================================
// 5. STRUCTURE (Pivots & MSS)
// =============================================================================

// Pivots
pl = ta.pivotlow(low, sw_left, sw_right)
ph = ta.pivothigh(high, sw_left, sw_right)

// Keep track of recent swings for MSS
var float last_swing_high = na
var float last_swing_low = na

if not na(ph)
    last_swing_high := high[sw_right]
if not na(pl)
    last_swing_low := low[sw_right]

// MSS Logic
// We look for MSS *after* a sweep has occurred
var bool mss_bull = false
var bool mss_bear = false

// Reset MSS flags when session restarts
if in_session and not in_session[1]
    mss_bull := false
    mss_bear := false

// Check MSS
if sweep_confirmed
    if swept_side == "Low" and not mss_bull
        // Looking for break of structure to upside
        if close > last_swing_high
            mss_bull := true
            if show_asia
                line.new(bar_index, last_swing_high, bar_index-1, last_swing_high, color=color.green, width=2)
                label.new(bar_index, last_swing_high, "MSS", style=label.style_none, textcolor=color.green, size=size.tiny)

    if swept_side == "High" and not mss_bear
        // Looking for break of structure to downside
        if close < last_swing_low
            mss_bear := true
            if show_asia
                line.new(bar_index, last_swing_low, bar_index-1, last_swing_low, color=color.red, width=2)
                label.new(bar_index, last_swing_low, "MSS", style=label.style_none, textcolor=color.red, size=size.tiny)

// =============================================================================
// 6. FVG & IFVG LOGIC
// =============================================================================

// 1. Detect New FVG
bool is_bull_fvg = (low[0] > high[2]) and (low[0] - high[2] > atrVal * fvg_threshold)
bool is_bear_fvg = (high[0] < low[2]) and (low[2] - high[0] > atrVal * fvg_threshold)

if is_bull_fvg
    Zone newZone = Zone.new(na, high[2], low[0], true, false, false, bar_index, true)
    if show_fvgs
        newZone.id := box.new(bar_index-2, newZone.top, bar_index + 5, newZone.btm, border_color=na, bgcolor=col_bull)
    array.push(zones, newZone)

if is_bear_fvg
    Zone newZone = Zone.new(na, low[2], high[0], false, false, false, bar_index, true)
    if show_fvgs
        newZone.id := box.new(bar_index-2, newZone.top, bar_index + 5, newZone.btm, border_color=na, bgcolor=col_bear)
    array.push(zones, newZone)

// 2. Update Zones (Mitigation & Inversion)
f_manage_zones() // Prune old

// Identify nearest valid zone for entry
var Zone valid_entry_zone = na

// Loop through zones to check status
if array.size(zones) > 0
    for i = 0 to array.size(zones) - 1
        Zone z = array.get(zones, i)

        if z.active
            // Extend visuals
            if not na(z.id)
                box.set_right(z.id, bar_index + 5)

            // Check Inversion (Close through zone)
            if not z.isInverted
                if z.isBull and close < z.btm
                    z.isInverted := true
                    z.isBull := false // Flips to supply
                    z.top := z.btm // Approximate flip area
                    z.btm := z.btm - (z.top - z.btm) // Visual adjust
                    if show_ifvgs
                        box.set_bgcolor(z.id, col_bear) // Turn red
                        box.set_border_color(z.id, color.red)
                else if not z.isBull and close > z.top
                    z.isInverted := true
                    z.isBull := true // Flips to demand
                    z.btm := z.top
                    z.top := z.top + (z.top - z.btm)
                    if show_ifvgs
                        box.set_bgcolor(z.id, col_bull) // Turn green
                        box.set_border_color(z.id, color.green)

            // Mitigation check (Price touching zone)
            // If inverted, we want retest from new side
            bool mitigated = false
            if z.isBull
                if low <= z.top and high >= z.btm
                    mitigated := true
            else
                if high >= z.btm and low <= z.top
                    mitigated := true

            if z.active
                valid_entry_zone := z

// =============================================================================
// 7. HTF BIAS
// =============================================================================

// Function to get HTF Structure
f_htf_struct() =>
    h = ta.pivothigh(high, 5, 5)
    l = ta.pivotlow(low, 5, 5)
    [h, l, close]

[h_htf, l_htf, c_htf] = request.security(syminfo.tickerid, tf_htf, f_htf_struct(), lookahead=barmerge.lookahead_off)

// Simple HTF Bias: Close vs recent Pivots
var string htf_bias = "Neutral"
var float htf_last_h = na
var float htf_last_l = na

if not na(h_htf)
    htf_last_h := h_htf
if not na(l_htf)
    htf_last_l := l_htf

if not na(c_htf) and not na(htf_last_h)
    if c_htf > htf_last_h
        htf_bias := "Bullish"
    else if not na(htf_last_l) and c_htf < htf_last_l
        htf_bias := "Bearish"

// Optional Daily Bias
[d_h, d_l, d_c] = request.security(syminfo.tickerid, tf_daily, f_htf_struct(), lookahead=barmerge.lookahead_off)
string daily_bias = "Neutral"
if use_daily
    // Simplified logic for daily
    daily_bias := d_c > d_c[1] ? "Bullish" : "Bearish"

// =============================================================================
// 8. SETUP TRIGGER & EXECUTION
// =============================================================================

bool setup_long = false
bool setup_short = false

// Trigger Conditions
// 1. Long
if mss_bull and not mss_bull[1] // Trigger on MSS candle close
    bool bias_ok = req_htf_bias ? (htf_bias == "Bullish" and (not use_daily or daily_bias == "Bullish")) : true
    if bias_ok
        setup_long := true

// 2. Short
if mss_bear and not mss_bear[1]
    bool bias_ok = req_htf_bias ? (htf_bias == "Bearish" and (not use_daily or daily_bias == "Bearish")) : true
    if bias_ok
        setup_short := true

// Entry Price Logic (Limit vs Market)
float nearby_bull_zone = na
float nearby_bear_zone = na

if array.size(zones) > 0
    for i = 0 to array.size(zones) - 1
        Zone z = array.get(zones, i)
        if z.active
            if z.isBull and close > z.btm
                nearby_bull_zone := z.top // Demand
            if not z.isBull and close < z.top
                nearby_bear_zone := z.btm // Supply

// Strategy Execution
if use_strat
    float limit_p = close
    float stop_price = na
    float tp_price = na
    float stop_dist = na
    float qty_pos = na

    // Long Execution
    if setup_long
        if not req_ifvg or not na(nearby_bull_zone)
            if entry_type == "Limit" and not na(nearby_bull_zone)
                // Determine limit level
                if limit_level == "Zone Start"
                    limit_p := nearby_bull_zone
                else if limit_level == "Zone Mid"
                    limit_p := nearby_bull_zone // fallback or math.avg(top,btm)

            stop_dist := (limit_p - asia_low)
            // Fallback SL if asia low is too far or na
            if na(stop_dist) or stop_dist <= 0
                stop_dist := atrVal * 2

            stop_price := math.min(limit_p - stop_dist, limit_p - (atrVal * sl_padding))
            tp_price := limit_p + (limit_p - stop_price) * rr_target

            if (limit_p - stop_price) > 0
                qty_pos := (strategy.equity * (risk_per_tr / 100)) / (limit_p - stop_price)

            strategy.entry("Long", strategy.long, qty=qty_pos, limit = entry_type == "Limit" ? limit_p : na)
            strategy.exit("Exit Long", "Long", stop=stop_price, limit=tp_price)

    // Short Execution
    if setup_short
        if not req_ifvg or not na(nearby_bear_zone)
            if entry_type == "Limit" and not na(nearby_bear_zone)
                if limit_level == "Zone Start"
                    limit_p := nearby_bear_zone

            stop_dist := (asia_high - limit_p)
            if na(stop_dist) or stop_dist <= 0
                stop_dist := atrVal * 2

            stop_price := math.max(limit_p + stop_dist, limit_p + (atrVal * sl_padding))
            tp_price := limit_p - (stop_price - limit_p) * rr_target

            if (stop_price - limit_p) > 0
                qty_pos := (strategy.equity * (risk_per_tr / 100)) / (stop_price - limit_p)

            strategy.entry("Short", strategy.short, qty=qty_pos, limit = entry_type == "Limit" ? limit_p : na)
            strategy.exit("Exit Short", "Short", stop=stop_price, limit=tp_price)

// =============================================================================
// 9. STATUS DASHBOARD
// =============================================================================

if show_table
    var tbl = table.new(position.top_right, 2, 6, bgcolor=color.new(color.black, 50), border_width=1)
    if barstate.islast
        table.cell(tbl, 0, 0, "HTF Bias (" + tf_htf + ")", text_color=color.white, text_size=size.small)
        table.cell(tbl, 1, 0, htf_bias, text_color = htf_bias == "Bullish" ? color.green : htf_bias == "Bearish" ? color.red : color.gray)

        table.cell(tbl, 0, 1, "Asia Range", text_color=color.white, text_size=size.small)
        table.cell(tbl, 1, 1, in_session ? "Active" : "Closed", text_color=color.yellow)

        table.cell(tbl, 0, 2, "Swept Side", text_color=color.white, text_size=size.small)
        table.cell(tbl, 1, 2, swept_side, text_color=color.white)

        table.cell(tbl, 0, 3, "MSS State", text_color=color.white, text_size=size.small)
        string mss_txt = mss_bull ? "Bull Confirmed" : mss_bear ? "Bear Confirmed" : "Waiting"
        table.cell(tbl, 1, 3, mss_txt, text_color= mss_bull ? color.green : mss_bear ? color.red : color.gray)

        table.cell(tbl, 0, 4, "Entry Signal", text_color=color.white, text_size=size.small)
        string sig_txt = setup_long ? "LONG SETUP" : setup_short ? "SHORT SETUP" : "Scanning..."
        table.cell(tbl, 1, 4, sig_txt, text_color=color.white, bgcolor = setup_long ? color.green : setup_short ? color.red : na)
